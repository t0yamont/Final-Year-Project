"""Dissertation figures from completed circuit baseline and PPO artifacts."""
import argparse,csv,gzip,json
from pathlib import Path
import numpy as np
import matplotlib
matplotlib.use('Agg')
import matplotlib.pyplot as plt
from benchmark_circuit_control import sha,save


def main():
    p=argparse.ArgumentParser(description=__doc__);p.add_argument('--pilot',type=Path,required=True);p.add_argument('--baselines',type=Path,required=True);p.add_argument('--output',type=Path,required=True);a=p.parse_args()
    inputs={}
    for folder in [a.pilot,a.baselines]:
        manifest=json.loads((folder/'manifest.json').read_text())
        for name,h in manifest['outputs'].items():
            if sha(name)!=h:raise ValueError('corrupt input: '+name)
        inputs[str(folder/'manifest.json')]=sha(folder/'manifest.json')
    metrics=json.loads((a.pilot/'metrics.json').read_text())
    with (a.pilot/'episodes.monitor.csv').open() as f:
        next(f);episodes=list(csv.DictReader(f))
    a.output.mkdir(parents=True,exist_ok=False)
    fig,axes=plt.subplots(1,2,figsize=(12,4.8))
    rewards=np.array([float(r['r']) for r in episodes]);steps=np.cumsum([int(r['l']) for r in episodes])
    axes[0].plot(steps,rewards,color='#176b8a',alpha=.4,label='Stochastic training episodes')
    if len(rewards)>=10:axes[0].plot(steps[9:],np.convolve(rewards,np.ones(10)/10,mode='valid'),color='#176b8a',label='10-episode mean')
    axes[0].set(xlabel='Completed training steps',ylabel='Episode reward = time saved (s)',title='Learning trace on training grid');axes[0].legend(fontsize=8)
    times=metrics['fine_baselines_s'];names=['half','full','spread','exit_boost','PPO']
    values=[times['zero']-times[n] for n in names[:-1]]+[metrics['fine_gain_over_zero_s']]
    axes[1].bar(names,values,color=['#777777']*4+['#176b8a'])
    for i,v in enumerate(values):axes[1].annotate(f'{v:.3f}',(i,v),xytext=(0,4),textcoords='offset points',ha='center',fontsize=9)
    axes[1].set(ylabel='Time saved versus zero request (s)',title='Deterministic fine-grid evaluation');axes[1].margins(y=.2)
    for ax in axes:ax.grid(axis='y',alpha=.2);ax.set_axisbelow(True)
    fig.suptitle(f'Full-circuit PPO pilot | {a.pilot.name} | seed {metrics["seed"]}, {metrics["actual_steps"]:,} steps')
    fig.text(.5,.01,'Assumed model, same55 decisions/lap and boundary conditions. One short seed; no convergence or held-out generalization claim.',ha='center',fontsize=9)
    fig.tight_layout(rect=(0,.05,1,.94))
    for ext in ['png','pdf']:fig.savefig(a.output/f'01_learning_and_comparison.{ext}',dpi=300)
    plt.close(fig)
    with gzip.open(a.pilot/'trained_fine_grid.json.gz','rt') as f:trained=json.load(f)
    profiles={'PPO':trained}
    for name in ['full','spread']:
        with gzip.open(a.baselines/f'blocks-55_cells-160_{name}.json.gz','rt') as f:profiles[name]=json.load(f)
    fig,axes=plt.subplots(3,1,figsize=(11,8),sharex=True)
    for (name,r),color,style in zip(profiles.items(),['#176b8a','#b86118','#555555'],['-','--',':']):
        trace=r['trace'];distance=np.array([0.]+[x['state']['distance_m'] for x in trace])/1000
        axes[0].stairs([x['action'] for x in trace],distance,baseline=None,label=name,color=color,linestyle=style)
        axes[1].plot(distance,[2.]+[x['state']['energy_j']/1e6 for x in trace],label=name,color=color,linestyle=style)
        axes[2].plot(distance,[70*3.6]+[x['state']['speed_m_per_s']*3.6 for x in trace],label=name,color=color,linestyle=style)
    axes[0].set(ylabel='Requested deployment fraction',ylim=(-.04,1.08));axes[0].legend(ncol=3,fontsize=9)
    axes[1].set(ylabel='Battery energy (MJ)');axes[1].axhline(.1,color='#999999',linewidth=.8)
    axes[2].set(ylabel='Speed (km/h)',xlabel='Distance from source origin (km)')
    for ax in axes:ax.grid(alpha=.2)
    fig.suptitle('Fine-grid policy replay at decision boundaries | '+a.pilot.name)
    fig.text(.5,.01,'Requests are held over each block; delivered power reduces during braking, cruise or depletion. Speed plotted at55 block endpoints.',ha='center',fontsize=9)
    fig.tight_layout(rect=(0,.04,1,.95))
    for ext in ['png','pdf']:fig.savefig(a.output/f'02_policy_energy_speed.{ext}',dpi=300)
    plt.close(fig)
    save(a.output/'manifest.json',dict(inputs=inputs,source_sha256=sha(__file__),outputs={str(x):sha(x) for x in a.output.iterdir() if x.is_file()}))


if __name__=='__main__':main()
